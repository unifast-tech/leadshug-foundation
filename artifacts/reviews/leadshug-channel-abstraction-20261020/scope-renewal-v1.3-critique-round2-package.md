  }

  async resolveForIngress(reference: VerifiedIngressChannelReference): Promise<ResolvedChannelOperation> {
    const conversation = await this.prisma.conversation.findFirst({
      where: { id: reference.conversationId, buId: reference.buId },
      select: { id: true, buId: true, contact: true },
    });
    if (!conversation) throw new ConflictException({ code: 'CHANNEL_CONTEXT_CHANGED' });
    const candidate = await this.findCandidate(reference.channelId, reference.buId);
    const identity = candidate?.providerIdentities.find((item) => item.id === reference.providerResourceIdentityId);
    if (!candidate || candidate.type !== reference.channelOrigin || candidate.generation !== reference.generation
      || candidate.pairingEpoch !== reference.pairingEpoch || !identity || identity.buId !== reference.buId
      || identity.provider !== reference.channelOrigin || identity.currentChannelId !== reference.channelId
      || !identity.managedLifecycle) {
      throw new ConflictException({ code: 'CHANNEL_CONTEXT_CHANGED' });
    }
    return this.bind(candidate);
  }

  async inspectForConversation(reference: ConversationChannelReference): Promise<ChannelAvailability> {
    try {
      const conversation = await this.prisma.conversation.findFirst({
        where: { id: reference.conversationId, buId: reference.buId }, select: { id: true },
      });
      if (!conversation) return this.unavailable(null, 'CANAL_SELECIONADO_INDISPONIVEL');
      const { candidate, selectionReason, selectedOrigin } = await this.selectCandidate(reference);
      if (!candidate) {
        return this.unavailable(selectedOrigin, selectionReason ?? 'CANAL_SELECIONADO_INDISPONIVEL');
      }
      const identity = candidate.providerIdentities.find((item) => item.currentChannelId === candidate.id);
      let reason: ChannelAvailability['reason'] = null;
      if (!candidate.ownNumberVerified || candidate.verifiedEpoch !== candidate.pairingEpoch || !identity
        || !identity.managedLifecycle || identity.provider !== candidate.type || identity.buId !== candidate.buId) {
        reason = 'CHANNEL_IDENTITY_NOT_VERIFIED';
      } else if (candidate.status !== 'CONECTADO' || !candidate.receiptVerifiedAt
        || (candidate.type === 'OFICIAL' && candidate.providerConnectionState !== 'SUBSCRIBED')
        || (candidate.type === 'NAO_OFICIAL' && candidate.providerConnectionState !== 'CONNECTED')
        || !this.adapters.locallyConfigured(candidate.type)) {
        reason = 'CHANNEL_NOT_READY';
      } else if (candidate.type === 'OFICIAL') {
        const credential = await this.credentials.inspectActiveCredential(candidate.id, candidate.generation);
        if (!credential.ready) reason = 'PENDING_CREDENTIAL';
      }

      const window = candidate.requiresWindow ? await this.window(reference.conversationId) : { open: true, expiresAt: null };
      if (!reason && candidate.requiresWindow && !window.open) reason = 'JANELA_FECHADA';
      return {
        available: reason === null,
        reason,
        channelOrigin: candidate.type,
        requiresWindow: candidate.requiresWindow,
        windowOpen: window.open,
        expiresAt: window.expiresAt,
      };
    } catch {
      return this.unavailable(null, 'CANAL_SELECIONADO_INDISPONIVEL');
    }
  }

  /** Batch-only GET view: BU bindings and credential metadata are read once for the page. */
  async inspectManyForConversations(references: readonly ConversationAvailabilityReference[]): Promise<Map<string, ChannelAvailability>> {
    if (!references.length) return new Map();
    const buId = references[0].buId;
    if (references.some((reference) => reference.buId !== buId)) {
      return new Map(references.map(({ conversationId }) => [conversationId, this.unavailable(null, 'CANAL_SELECIONADO_INDISPONIVEL')]));
    }
    try {
      const ids = references.map(({ conversationId }) => conversationId);
      const channels = await this.channels(buId);
      const credentialState = await this.credentials.inspectActiveCredentials(channels
        .filter((candidate) => candidate.type === 'OFICIAL')
        .map(({ id, generation }) => ({ channelId: id, generation })));
      const selectedById = new Map<string, {
        candidate: Candidate | null;
        selectedOrigin: ChannelOrigin | null;
        selectionReason: ChannelAvailability['reason'] | null;
      }>();

        let selectedOrigin = reference.latestMessageOrigin;
        let candidate: Candidate | null;
        let selectionReason: ChannelAvailability['reason'] | null = null;
        if (selectedOrigin) {
          candidate = channels.find((item) => item.type === selectedOrigin) ?? null;
          if (!candidate) selectionReason = 'CANAL_SELECIONADO_INDISPONIVEL';
        } else if (channels.length === 0) {
          candidate = null; selectionReason = 'CANAL_AUSENTE';
        } else if (channels.length > 1) {

        } else {

        }

      }

        .filter(({ conversationId }) => selectedById.get(conversationId)?.candidate?.requiresWindow)

          continue;
        }
        const identity = candidate.providerIdentities.find((item) => item.currentChannelId === candidate.id);
        let reason: ChannelAvailability['reason'] = null;
        if (!candidate.ownNumberVerified || candidate.verifiedEpoch !== candidate.pairingEpoch || !identity
          || !identity.managedLifecycle || identity.provider !== candidate.type || identity.buId !== candidate.buId) {
          reason = 'CHANNEL_IDENTITY_NOT_VERIFIED';
        } else if (candidate.status !== 'CONECTADO' || !candidate.receiptVerifiedAt
          || (candidate.type === 'OFICIAL' && candidate.providerConnectionState !== 'SUBSCRIBED')
          || (candidate.type === 'NAO_OFICIAL' && candidate.providerConnectionState !== 'CONNECTED')
          || !this.adapters.locallyConfigured(candidate.type)) {
          reason = 'CHANNEL_NOT_READY';
        } else if (candidate.type === 'OFICIAL' && !credentialState.get(candidate.id)?.ready) {
          reason = 'PENDING_CREDENTIAL';
        }
        let windowOpen = true;
        let expiresAt: Date | null = null;
        if (candidate.requiresWindow) {
          const lastInbound = windows.get(id);
          windowOpen = Boolean(lastInbound && lastInbound.getTime() + 24 * 60 * 60 * 1000 > Date.now());
          expiresAt = lastInbound ? new Date(lastInbound.getTime() + 24 * 60 * 60 * 1000) : null;
          if (!reason && !windowOpen) reason = 'JANELA_FECHADA';
        }
        result.set(id, {
          available: reason === null, reason, channelOrigin: candidate.type,
          requiresWindow: candidate.requiresWindow, windowOpen, expiresAt,
        });
      }
      return result;
    } catch {
      return new Map(references.map(({ conversationId }) => [conversationId, this.unavailable(null, 'CANAL_SELECIONADO_INDISPONIVEL')]));
    }
  }

  async contextAfterDispatch(context: ChannelOperationContext): Promise<'CURRENT' | 'CHANGED' | 'UNKNOWN'> {
    try {
      const status = await this.contextStatus(context, false);
      return status;
    } catch {
      return 'UNKNOWN';
    }
  }

  private async bind(candidate: Candidate): Promise<ResolvedChannelOperation> {
    const availability = await this.inspectCandidate(candidate);
    if (availability) this.throwUnavailable(availability);
    const credential = await this.credentials.resolve(candidate.id);
    const context: ChannelOperationContext = Object.freeze({
      channelId: candidate.id,
      buId: candidate.buId,
      channelOrigin: candidate.type,
      providerResourceIdentityId: credential.providerResourceIdentityId,
      generation: credential.generation,
      pairingEpoch: credential.pairingEpoch,
      activeCredentialVersion: credential.activeCredentialVersion,
      requiresWindow: candidate.requiresWindow,
    });
    if (!this.sameCredentialContext(candidate, credential)) {
      throw new ConflictException({ code: 'CHANNEL_CONTEXT_CHANGED' });
    }
    const factory = this.adapters.get(candidate.type);
    if (!factory) throw new ConflictException({ code: 'CHANNEL_NOT_READY' });
    const channel = factory.create(credential, async () => {
      try { return (await this.contextStatus(context, true)) === 'CURRENT'; }
      catch { return false; }
    });
    return { channel, context };
  }

  private async inspectCandidate(candidate: Candidate): Promise<ChannelAvailability['reason']> {
    const identity = candidate.providerIdentities.find((item) => item.currentChannelId === candidate.id);
    if (!candidate.ownNumberVerified || candidate.verifiedEpoch !== candidate.pairingEpoch || !identity
      || !identity.managedLifecycle || identity.provider !== candidate.type || identity.buId !== candidate.buId) {
      return 'CHANNEL_IDENTITY_NOT_VERIFIED';
    }
    if (candidate.status !== 'CONECTADO' || !candidate.receiptVerifiedAt
      || (candidate.type === 'OFICIAL' && candidate.providerConnectionState !== 'SUBSCRIBED')
      || (candidate.type === 'NAO_OFICIAL' && candidate.providerConnectionState !== 'CONNECTED')
      || !this.adapters.locallyConfigured(candidate.type)) return 'CHANNEL_NOT_READY';
    if (candidate.type === 'OFICIAL' && !(await this.credentials.inspectActiveCredential(candidate.id, candidate.generation)).ready) {
      return 'PENDING_CREDENTIAL';
    }
    return null;
  }

  private async selectForConversation(reference: ConversationChannelReference) {
    const conversation = await this.prisma.conversation.findFirst({
      where: { id: reference.conversationId, buId: reference.buId },
      select: { id: true, buId: true, contact: true },
    });
    if (!conversation) throw new ConflictException({ code: 'CONVERSATION_IDENTITY_MISMATCH' });
    const { candidate, selectionReason } = await this.selectCandidate(reference);
    if (!candidate) this.throwUnavailable(selectionReason ?? 'CANAL_AUSENTE');
    const reason = await this.inspectCandidate(candidate);
    if (reason) this.throwUnavailable(reason);
    if (candidate.requiresWindow) {
      const window = await this.window(conversation.id);
      if (!window.open) this.throwUnavailable('JANELA_FECHADA');
    }
    return { conversation, candidate };
  }

  private async selectCandidate(reference: ConversationChannelReference): Promise<{
    candidate: Candidate | null; selectedOrigin: ChannelOrigin | null; selectionReason: ChannelAvailability['reason'] | null;
  }> {
    const latest = await this.prisma.message.findFirst({
      where: { conversationId: reference.conversationId },
      orderBy: [{ createdAt: 'desc' }, { id: 'desc' }],
      select: { channelOrigin: true },
    });
    const channels = await this.channels(reference.buId);
    if (latest) {
      const selectedOrigin = latest.channelOrigin as ChannelOrigin;
      const candidate = channels.find((channel) => channel.type === selectedOrigin) ?? null;
      return { candidate, selectedOrigin, selectionReason: candidate ? null : 'CANAL_SELECIONADO_INDISPONIVEL' };
    }
    if (channels.length === 0) return { candidate: null, selectedOrigin: null, selectionReason: 'CANAL_AUSENTE' };
    if (channels.length > 1) return { candidate: null, selectedOrigin: null, selectionReason: 'CANAL_AMBIGUO' };
    return { candidate: channels[0], selectedOrigin: channels[0].type, selectionReason: null };
  }

  private async channels(buId: string): Promise<Candidate[]> {
    return this.prisma.buChannel.findMany({
      where: { buId },
      select: {

      },

  }

  private async findCandidate(channelId: string, buId: string): Promise<Candidate | null> {

    } });
    return (rows[0] as unknown as Candidate | undefined) ?? null;
  }

  private async contextStatus(context: ChannelOperationContext, beforeDispatch: boolean): Promise<'CURRENT' | 'CHANGED' | 'UNKNOWN'> {
    const where: Prisma.BuChannelWhereInput = {
      id: context.channelId, buId: context.buId, type: context.channelOrigin,
      generation: context.generation, pairingEpoch: context.pairingEpoch,
      ownNumberVerified: true, verifiedEpoch: context.pairingEpoch,
      ...(beforeDispatch ? {
        status: 'CONECTADO', receiptVerifiedAt: { not: null },
        providerConnectionState: context.channelOrigin === 'OFICIAL' ? 'SUBSCRIBED' : 'CONNECTED',
      } : {}),
    };
    const channel = await this.prisma.buChannel.findFirst({
      where,
      select: { id: true, credential: { select: { version: true, generation: true } },
        providerIdentities: { where: { id: context.providerResourceIdentityId }, select: { id: true, buId: true, provider: true, currentChannelId: true, managedLifecycle: true } } },
    });
    const identity = channel?.providerIdentities[0];
    if (!channel || !identity || identity.buId !== context.buId || identity.provider !== context.channelOrigin
      || identity.currentChannelId !== context.channelId || !identity.managedLifecycle) {
      return 'CHANGED';
    }
    if (context.channelOrigin === 'OFICIAL') {
      const credential = channel.credential;
      if (!credential || credential.generation !== context.generation) return 'CHANGED';
      if (beforeDispatch && credential.version !== context.activeCredentialVersion) return 'CHANGED';
    }
    return 'CURRENT';
  }

  private async window(conversationId: string): Promise<{ open: boolean; expiresAt: Date | null }> {
    const lastInbound = await this.prisma.message.findFirst({
      where: { conversationId, direction: 'IN' }, orderBy: [{ createdAt: 'desc' }, { id: 'desc' }], select: { createdAt: true },
    });
    if (!lastInbound) return { open: false, expiresAt: null };
    const expiresAt = new Date(lastInbound.createdAt.getTime() + 24 * 60 * 60 * 1000);
    return { open: expiresAt.getTime() > Date.now(), expiresAt };
  }

  private async latestInboundByConversation(conversationIds: string[]): Promise<Map<string, Date>> {
    if (!conversationIds.length) return new Map();
    const rows = await this.prisma.$queryRaw<Array<{ conversation_id: string; created_at: Date }>>`
      SELECT requested.conversation_id, latest.created_at
      FROM unnest(ARRAY[${Prisma.join(conversationIds)}]::text[]) AS requested(conversation_id)
      CROSS JOIN LATERAL (
        SELECT message.created_at
        FROM messages AS message
        WHERE message.conversation_id = requested.conversation_id AND message.direction = 'IN'
        ORDER BY message.created_at DESC, message.id DESC
        LIMIT 1
      ) AS latest
    `;
    return new Map(rows.map((row) => [row.conversation_id, row.created_at]));
  }

  private sameCredentialContext(candidate: Candidate, credential: ResolvedChannelCredential): boolean {
    const identity = candidate.providerIdentities.find((item) => item.id === credential.providerResourceIdentityId);
    return credential.channelId === candidate.id && credential.buId === candidate.buId
      && credential.provider === candidate.type && credential.generation === candidate.generation
      && credential.pairingEpoch === candidate.pairingEpoch && identity?.currentChannelId === candidate.id
      && identity?.buId === candidate.buId && identity.provider === candidate.type;
  }

  private unavailable(channelOrigin: ChannelOrigin | null, reason: ChannelAvailability['reason']): ChannelAvailability {
    return { available: false, reason, channelOrigin, requiresWindow: false, windowOpen: false, expiresAt: null };
  }

  private throwUnavailable(reason: ChannelAvailability['reason']): never {
    if (reason === 'CANAL_AUSENTE') throw new BadRequestException({ code: reason, error: 'Esta BU não tem canal conectado.' });
    if (reason === 'CANAL_AMBIGUO') throw new ConflictException({ code: reason, error: 'Selecione uma origem de conversa antes de responder.' });
    throw new ConflictException({ code: reason, error: reason === 'JANELA_FECHADA'
      ? 'Fora da janela de 24h da última mensagem recebida.'
      : 'O canal selecionado não está disponível para envio.' });
  }
}

# CI Equivalent Governance

Use this skill whenever a workflow, skill, TODO, execution plan, or promotion lane says `CI Equivalent`, claims `stage-full`-style parity, or needs to decide whether local evidence is truly pipeline-equivalent instead of merely diagnostic.

## Purpose
Freeze the meaning of `CI Equivalent` as current-branch local product proof on the authoritative branch under evaluation. This skill owns the shared taxonomy so promotion, test orchestration, TODO delivery, pipeline updates, and reconciliation do not redefine it differently.

## Canonical Model
- `CI Equivalent` is local proof, not published-lane proof.
- It runs on the current authoritative branch under evaluation (`feature/*`, `review/*`, `sequence/*`, `reconcile/*`, version/package branch such as `v0.2.0+8-rc`, or equivalent).
- It uses the project-owned local build/publish/runtime path and the same product-facing suites/jobs the pipeline uses for that scope on that branch.
- It is scope-complete for the touched repo/job family. A targeted rerun, representative smoke path, or single happy path is diagnostic evidence unless the approved matrix explicitly says that narrower scope is complete.
- Published `stage` or `main` probes are separate evidence. They never replace `CI Equivalent`.
- `CI Equivalent` is a generic proof concept. Reconciliation is only one possible execution topology.
- If the required local runtime surface for a broad gate is unavailable, hung, or ambiguous, the `CI Equivalent` claim is `blocked`, not `passed`, and published-lane evidence still does not replace it.

## Authoritative Branch And Wrapper Discipline
- The branch under test is the current authoritative branch for the claim being made.
- For package/version promotion lanes, the authoritative source branch is the exact branch recorded in the governing TODO `Current Branch Authority`, typically the version `*-rc` branch. Example: `v0.2.0+8-rc`.
- In those package/version lanes, the first promotion-grade `CI Equivalent` run must pass on that authoritative `*-rc` branch before any derived `review/*` remediation branch is opened.
- Reconcile-only wrappers are valid only on real `reconcile/*` states or an explicitly recorded same-commit reconcile alias.
- In real orchestration, the consolidated `reconcile/*` branch is authoritative until green; after that, replay the accepted net effect onto the recorded canonical return branch before promotion or non-orchestration closeout resumes.
- In TODO sequencing, a `sequence/*` branch may be authoritative for the one integrated package-closeout broad local gate only when that branch is the current principal-checkout authoritative branch under evaluation and owns the real local runtime/browser surface. Individual TODO checkpoints use their recorded affected-area bundles and must not invoke or claim the broad gate by default. The sequencing branch is checkpoint-only topology and is never itself a promotable source branch.
- If `sequence/*` exists only inside an isolated linked worktree while the principal checkout is reserved for another lane, treat it as non-authoritative implementation topology only: it may run an explicitly narrower checkpoint prefix gate, but it must not run or claim the parity-complete broad local gate until the accepted net effect is replayed onto the principal authoritative branch.
- Review/remediation branches may run `CI Equivalent` for their own validation loop, but they are evidence branches rather than promotable source authority.
- A `review/*` pass never substitutes for the missing authoritative `*-rc` pass. It is an additional gate.
- After replay from `review/*` back onto the authoritative `*-rc` branch, rerun `CI Equivalent` there whenever the replay was not a pure fast-forward or otherwise introduced branch-local change.

## Broad Local Stage Gate Rule
When a repo exposes a named broad local stage gate such as `stage-full`, that name is reserved for the parity-complete local mirror of the stage pipeline for the touched scope on that branch.

That parity must include both:
- the complete repo-owned suite/job family that the stage pipeline executes for that scope; and
- any pipeline-owned lifecycle work outside the inner smoke commands, such as build/publish/provenance, fixture/bootstrap seed, host-override management, cleanup/teardown, restore/readback preparation, or equivalent pre/post steps.

A local bundle is not valid `stage-full` parity when any of these are true:
- it omits an in-scope stage suite/job that the pipeline still runs for that scope;
- it runs only inner smoke commands while the pipeline owns required outer lifecycle steps;
- it passes only because prior state already had healthy fixtures, seed data, runtime overrides, or published artifacts;
- it silently changes branch/topology semantics just to satisfy a reconcile-only wrapper.

Narrower diagnostic bundles must use distinct names and must not be reported as the broad stage gate.

If an isolated sequencing worktree uses a prefix derived from `stage-full`, that prefix must stop before `browser-stage-full`. That cutoff means the isolated state never enters `local-public-web-build`, readonly smoke, or mutation smoke, and it must not claim any of those surfaces. Any request to cross that cutoff from the isolated worktree is out of contract and must be refused until replay onto the principal authoritative branch.

When a broad local gate such as `stage-full` cannot complete because the local runtime surface is unhealthy or non-responsive:
- treat the gate as a blocking local infrastructure failure, not as a soft warning;
- do not continue promotion, replay, or closeout on the theory that remote `stage` or `main` evidence is "good enough";
- do not reinterpret targeted local checks as a substitute for the missing broad gate;
- only explicit human waiver may authorize progression without that gate;
- emit a PACED/TEACH stop response that names the blocked gate, the failing runtime surface, why evolution is disallowed, and the next local recovery action.

## Mutation And Production Semantics
- Mutation validation belongs only on an approved non-`main` lane.
- If the touched behavior includes mutation, `CI Equivalent` must include corresponding local mutation evidence on that non-production lane.
- Production/main semantics may intentionally forbid mutation. Prove that separately with a production-lane proof surface such as `main-proof`; do not weaken non-main `CI Equivalent` to avoid running mutation locally.

## Required Evidence Hygiene
Before claiming `CI Equivalent`, record:
- authoritative branch under evaluation;
- governing TODO branch authority when package/version promotion is in scope, including the exact `*-rc` branch name and validated `branch@sha` when applicable;
- exact local commands or contract profiles run;
- runtime freshness attestation for every manual/browser/device proof surface: authoritative `branch@sha`, local build/publish artifact or fingerprint, served target URL/device/tunnel, and the comparison/probe proving the served target matched that build before the test result was interpreted;
- touched repo/job family covered;
- runtime topology, publish path, domains, tenants, and credentials lane when relevant;
- whether readonly and mutation behaviors were both required;
- whether pipeline-owned lifecycle steps were executed locally as part of the same contract;
- evidence artifacts proving the current build/branch state was the one actually served.

Browser/device evidence is invalid if it cannot prove the current build, bundle, or runtime state was what the runner exercised. Freshness must be proven before the test result is trusted, not reconstructed afterward from assumption.

## Promotable Documentation Preflight
For a promotable broad gate, validate required delivery documentation before starting the expensive execution contract. This is a precondition of execution, not a second kind of post-execution CI-equivalent proof.

- The preflight must fail closed when a TODO or delivery record that should already describe completed implementation/evidence remains `Pending`, `Open`, or otherwise stale.
- The preflight must allow the one explicitly declared pending item that the current broad gate itself will produce (for example, the pending `promotable-stage-full` evidence row). It must not require future execution evidence before the execution that creates it.
- Correcting a scope-neutral documentation drift found by this preflight happens before `stage-full`; after the correction, run the preflight again and then run `stage-full` once on the final documented state.
- Do not make a completed `stage-full` rerun merely because a later promotion guard exposed a scope-neutral documentation omission that the preflight should have caught. Harden the preflight and correct the documentation instead.
- If the documentation correction changes declared scope, behavior, acceptance criteria, runtime contract, or another execution-relevant claim, it is no longer documentation-only: reassess the execution evidence and rerun the broad gate when the project-owned invalidation policy requires it.

## Evidence Reuse And Invalidation
- Exact `branch@sha` remains the traceability anchor for CI-equivalent and promotion claims. Do not stop recording it.
- SHA drift alone is not the canonical rerun trigger once a passed CI-equivalent artifact already exists.
- Reuse is allowed only when a deterministic, project-owned invalidation/reuse guard proves the current authoritative branch differs from that passed baseline only on explicitly safe-reuse surfaces.
- Rerun is required when the guard reports any invalidating drift in product code, test ownership, wrappers, lifecycle steps, build/publish/provenance paths, runtime/bootstrap/config, or other stage-facing surfaces.
- Manual admission/authority refresh is required when the current branch is dirty, diverged from the baseline topology, or the baseline artifact/TODO/policy no longer line up deterministically.
- Backward-compatibility note: if an older report artifact lacks recorded repo head metadata, the guard may fall back to the governing TODO `branch@sha` entries, but that is weaker than artifact-native repo-state evidence and should be phased out.

- Run this method whenever `wf-docker-audit-escalation-method` marks `critique` as `required|recommended`.

- The audit-escalation guard decides whether the package depth stays `baseline` or becomes `expanded`.
